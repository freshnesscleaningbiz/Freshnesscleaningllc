            document.querySelectorAll('#bath-options button').forEach((btn, idx) => {
                if (idx === num - 1) {
                    btn.classList.add('bg-fresh-500', 'text-white', 'border-fresh-500');
                } else {
                    btn.classList.remove('bg-fresh-500', 'text-white', 'border-fresh-500');
                }
            });
            calculatePrice();
        }

        function setServiceType(type) {
            selectedServiceType = type;
            ['standard', 'deep', 'movein'].forEach(t => {
                const el = document.getElementById('type-' + t);
                if (t === type) {
                    el.classList.add('border-fresh-500', 'bg-fresh-50/50');
                    el.classList.remove('border-slate-200');
                } else {
                    el.classList.remove('border-fresh-500', 'bg-fresh-50/50');
                    el.classList.add('border-slate-200');
                }
            });
            calculatePrice();
        }

        function setFreq(mult, name) {
            resFreqMultiplier = mult;
            const buttons = document.querySelectorAll('.freq-btn');
            buttons.forEach(btn => {
                if (btn.textContent.includes(name)) {
                    btn.classList.add('border-fresh-500', 'bg-fresh-500', 'text-white');
                } else {
                    btn.classList.remove('border-fresh-500', 'bg-fresh-500', 'text-white');
                }
            });
            calculatePrice();
        }

        function updateSqft(val) {
            commercialSqft = parseInt(val);
            document.getElementById('sqft-display').textContent = Number(val).toLocaleString() + ' Sq Ft';
            calculatePrice();
        }

        function setCommFreq(mult, name) {
            commFreqMultiplier = mult;
            const buttons = document.querySelectorAll('.cfreq-btn');
            buttons.forEach(btn => {
                if (btn.textContent.includes(name)) {
                    btn.classList.add('border-fresh-500', 'bg-fresh-500', 'text-white');
                } else {
                    btn.classList.remove('border-fresh-500', 'bg-fresh-500', 'text-white');
                }
            });
            calculatePrice();
        }

        function calculatePrice() {
            let total = 0;
            if (currentMode === 'residential') {
                let base = 80;
                let bedFee = (selectedBeds - 1) * 20;
                let bathFee = (selectedBaths - 1) * 25;
                
                let typeMultiplier = 1.0;
                if (selectedServiceType === 'deep') typeMultiplier = 1.5;
                if (selectedServiceType === 'movein') typeMultiplier = 1.8;

                let addOnTotal = 0;
                if (document.getElementById('add-oven').checked) addOnTotal += 35;
                if (document.getElementById('add-fridge').checked) addOnTotal += 35;
                if (document.getElementById('add-windows').checked) addOnTotal += 45;

                total = ((base + bedFee + bathFee) * typeMultiplier + addOnTotal) * resFreqMultiplier;
                document.getElementById('estimated-unit').textContent = '/ service';
            } else {
                // Commercial Mode Calculation: $120 up to 1,500 sq ft; $220 for 1,500-3,500 sq ft; custom scaling above 3,500
                if (commercialSqft <= 1500) {
                    total = 120 * commFreqMultiplier;
                } else if (commercialSqft <= 3500) {
                    total = 220 * commFreqMultiplier;
                } else {
                    let base220 = 220;
                    let extraSqft = commercialSqft - 3500;
                    total = (base220 + Math.floor(extraSqft / 500) * 30) * commFreqMultiplier;
                }
                document.getElementById('estimated-unit').textContent = '/ visit';
            }

            const formattedTotal = '$' + Math.round(total);
            document.getElementById('estimated-total').textContent = formattedTotal;

            // Sync with hidden fields in request form
            const unit = document.getElementById('estimated-unit').textContent;
            document.getElementById('form-calc-estimate').value = `${formattedTotal} ${unit}`;
        }

        function bookThisClean() {
            // Fill size details in contact form
            const sizeDetailsInput = document.getElementById('form-size-details');
            const serviceTypeSelect = document.getElementById('form-service-type');

            if (currentMode === 'residential') {
                sizeDetailsInput.value = `${selectedBeds} Bed, ${selectedBaths} Bath (${selectedServiceType.toUpperCase()})`;
                if (selectedServiceType === 'standard') serviceTypeSelect.value = 'Residential Standard Clean';
                if (selectedServiceType === 'deep') serviceTypeSelect.value = 'Residential Deep Clean';
                if (selectedServiceType === 'movein') serviceTypeSelect.value = 'Residential Move-In/Out Clean';
            } else {
                sizeDetailsInput.value = `Commercial Space - ${Number(commercialSqft).toLocaleString()} Sq Ft`;
                serviceTypeSelect.value = 'Commercial Office Cleaning';
            }

            // Scroll down to contact form smoothly
            document.getElementById('contact').scrollIntoView({ behavior: 'smooth' });
        }

        // Initialize default calculation on load
        window.onload = function() {
            calculatePrice();
        };
    </script>
